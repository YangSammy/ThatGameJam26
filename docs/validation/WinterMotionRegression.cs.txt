var f=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;var ty=typeof(MemoryWhitebox.PaintingChapter);var step=ty.GetMethod("Step",f);var log=new System.Collections.Generic.List<string>();
System.Func<int,MemoryWhitebox.PaintingChapter> make=phase=>{var s=new MemoryWhitebox.PaintingChapter();ty.GetProperty("Phase").GetSetMethod(true).Invoke(s,new object[]{phase});return s;};
System.Action<bool,string> check=(ok,label)=>{if(!ok)throw new Exception(label);log.Add("PASS "+label);};
foreach(float side in new[]{-1f,1f})foreach(float axis in new[]{-1f,0f,1f}){var s=make(1);s.Painted[2]=true;s.CrateX=2;s.Actor=new Vector2(2+side*.60f,0);step.Invoke(s,new object[]{.01f,axis,false});check((s.Actor.x-s.CrateX)*side>0,"crate side "+side+" axis "+axis);check(axis*side<0?Mathf.Abs(s.CrateX-(2+axis*.025f))<.0001f:s.CrateX==2,"push only toward crate");}
foreach(int phase in new[]{0,1})foreach(float vy in new[]{5f,-1f}){var s=make(phase);s.Painted[phase==0?0:3]=true;var seat=phase==0?s.SwingSeat:s.FlowerSeat(0);s.Actor=seat+Vector2.up*.2f;s.Grounded=false;s.Velocity=new Vector2(0,vy);step.Invoke(s,new object[]{.01f,0f,false});check(vy>0?s.Riding==-1:s.Riding>=0,"phase "+phase+" vertical speed "+vy);}
var under=make(0);under.Painted[0]=true;under.Actor=under.SwingSeat-Vector2.up*.2f;under.Grounded=false;under.Velocity=Vector2.down;step.Invoke(under,new object[]{.01f,0f,false});check(under.Riding==-1,"no airborne boarding from below");
var ground=make(0);ground.Painted[0]=true;ground.SwingAngle=0;ground.Actor=new Vector2(ground.SwingSeat.x,ground.SwingSeat.y+.01f);ground.Grounded=false;ground.Velocity=Vector2.down;step.Invoke(ground,new object[]{.01f,0f,false});check(ground.Riding==0,"swing landing remains available");
for(int ride=0;ride<=6;ride++){var s=make(ride==0?0:1);s.Riding=ride;s.SwingSpeed=3;step.Invoke(s,new object[]{0f,1f,true});check(s.Riding==-1 && s.Velocity==new Vector2(2.5f,5),"normal jump from ride "+ride);}
return string.Join("\n",log.ToArray());
